    let currentIndex = 0;
    let score = 0;
    let isProcessing = false;

    // DOM 요소
    const gameScreen = document.getElementById('game-screen');
    const clearScreen = document.getElementById('clear-screen');
    const questionCountEl = document.getElementById('question-count');
    const scoreCountEl = document.getElementById('score-count');
    const progressBarEl = document.getElementById('progress-bar');
    const wordCardEl = document.getElementById('word-card');
    const typeTagEl = document.getElementById('type-tag');
    const cardWordEl = document.getElementById('card-word');
    const cardHintEl = document.getElementById('card-hint');
    const userInputEl = document.getElementById('user-input');
    const submitBtn = document.getElementById('submit-btn');
    const messageEl = document.getElementById('message');
    const restartBtn = document.getElementById('restart-btn');

    // 화면 업데이트
    function loadQuestion() {
      const q = quizData[currentIndex];
      typeTagEl.textContent = q.type;
      typeTagEl.className = `type-tag ${q.tagClass}`;
      cardWordEl.textContent = q.word;
      cardHintEl.textContent = q.hint;
      
      questionCountEl.textContent = `문제 ${currentIndex + 1} / ${quizData.length}`;
      scoreCountEl.textContent = `점수: ${score}점`;
      progressBarEl.style.width = `${((currentIndex + 1) / quizData.length) * 100}%`;
      
      userInputEl.value = '';
      userInputEl.focus();
      messageEl.textContent = '';
      messageEl.className = 'message-area';
      wordCardEl.className = 'word-card';
      isProcessing = false;
    }

    // 정답 제출 확인
    function checkAnswer() {
      if (isProcessing) return;

      const inputVal = userInputEl.value.trim();
      if (!inputVal) {
        showMessage('단어를 입력해 주세요!', 'msg-incorrect');
        return;
      }

      const q = quizData[currentIndex];
      const isCorrect = q.validAnswers.includes(inputVal);

      if (isCorrect) {
        isProcessing = true;
        score += 10;
        scoreCountEl.textContent = `점수: ${score}점`;
        
        wordCardEl.classList.add('correct-anim');
        showMessage('정답! 🎉 (+10점)', 'msg-correct');

        setTimeout(() => {
          currentIndex++;
          if (currentIndex < quizData.length) {
            loadQuestion();
          } else {
            showClearScreen();
          }
        }, 1200);

      } else {
        wordCardEl.classList.add('incorrect-anim');
        showMessage('다시 해보세요! (사전에 없거나 짜임에 맞지 않는 단어예요)', 'msg-incorrect');

        setTimeout(() => {
          wordCardEl.classList.remove('incorrect-anim');
        }, 400);
      }
    }

    function showMessage(text, className) {
      messageEl.textContent = text;
      messageEl.className = `message-area ${className}`;
    }

    function showClearScreen() {
      gameScreen.style.display = 'none';
      clearScreen.style.display = 'block';
    }

    function resetGame() {
      currentIndex = 0;
      score = 0;
      clearScreen.style.display = 'none';
      gameScreen.style.display = 'block';
      loadQuestion();
    }

    // 이벤트 리스너
    submitBtn.addEventListener('click', checkAnswer);
    userInputEl.addEventListener('keyup', (e) => {
      if (e.key === 'Enter') checkAnswer();
    });
    restartBtn.addEventListener('click', resetGame);

    // 최초 게임 시작
    loadQuestion();
  </script>
